cmake_minimum_required(VERSION 3.16)
project(gxde-lock-classic LANGUAGES CXX)

include(GNUInstallDirs)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)

# The top level enables these Qt porting guards for all of its children.
# The DTK widgets still use implicit QString conversions and Qt's foreach.
remove_definitions(
    -DQT_NO_CAST_FROM_ASCII
    -DQT_NO_CAST_FROM_BYTEARRAY
    -DQT_NO_FOREACH
)

find_package(Qt6 COMPONENTS Core Gui Widgets DBus Network Xml Svg Qml WaylandClient REQUIRED)
if(NOT TARGET Qt6::WaylandClientPrivate)
    find_package(Qt6 COMPONENTS WaylandClientPrivate REQUIRED)
endif()
find_package(LayerShellQt REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(DTK2WIDGET REQUIRED IMPORTED_TARGET dtk2widget)
pkg_check_modules(GSETTINGS_QT6 REQUIRED IMPORTED_TARGET gsettings-qt6)
pkg_check_modules(WAYLAND_PROTOCOLS REQUIRED wayland-protocols)
pkg_get_variable(WAYLAND_PROTOCOLS_DATADIR wayland-protocols pkgdatadir)
find_program(WAYLAND_SCANNER_EXECUTABLE NAMES wayland-scanner REQUIRED)

# Repo root: the lock reuses the SDDM authenticator (src/auth + src/common).
get_filename_component(LOCK_REPO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)

# --- Widget sources ----------------------------------------------------------
file(GLOB NEO_COMMON_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util/dbus/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/widgets/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/session_widgets/*.cpp"
)
# Wallpaper, cursor, scale and OS defaults, shared with gxde-lock-neo.
list(APPEND NEO_COMMON_SOURCES
    "${LOCK_REPO_ROOT}/src/common/LockAppearance.cpp"
    "${LOCK_REPO_ROOT}/src/common/LockCursor.cpp"
    "${LOCK_REPO_ROOT}/src/common/LockScale.cpp"
    "${LOCK_REPO_ROOT}/src/common/SystemDefaults.cpp"
)
file(GLOB NEO_COMMON_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util/dbus/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/widgets/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/session_widgets/*.h"
)
list(APPEND NEO_COMMON_HEADERS
    "${LOCK_REPO_ROOT}/src/common/LockAppearance.h"
    "${LOCK_REPO_ROOT}/src/common/LockCursor.h"
    "${LOCK_REPO_ROOT}/src/common/LockScale.h"
    "${LOCK_REPO_ROOT}/src/common/SystemDefaults.h"
)

set(NEO_VENDOR_SOURCES
    vendor/src/backend/users/users.cc
    vendor/src/backend/sessions/sessions.cc
    vendor/src/utils/logger/logger.cc
)
set(NEO_VENDOR_HEADERS
    vendor/src/backend/users/users.h
    vendor/src/backend/sessions/sessions.h
    vendor/src/backend/global_def/global.h
    vendor/src/utils/logger/logger.h
    vendor/src/utils/logger/ansi_escape_code_def.h
)

# Resources used by shared widgets (compiled into each executable).
set(NEO_WIDGET_RESOURCES
    widgets/widgetsimages.qrc
    widgets/widgetstheme.qrc
)

set(NEO_INCLUDE_DIRS
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util"
    "${CMAKE_CURRENT_SOURCE_DIR}/global_util/dbus"
    "${CMAKE_CURRENT_SOURCE_DIR}/widgets"
    "${CMAKE_CURRENT_SOURCE_DIR}/session_widgets"
    "${CMAKE_CURRENT_SOURCE_DIR}/vendor"
    "${LOCK_REPO_ROOT}/src/common"
)

set(NEO_COMMON_LIBS
    Qt6::Core Qt6::Gui Qt6::Widgets Qt6::DBus Qt6::Network Qt6::Xml Qt6::Svg
    PkgConfig::DTK2WIDGET
    PkgConfig::GSETTINGS_QT6
    X11 Xi Xtst Xcursor Xfixes Xrandr Xext
)

# --- SDDM authenticator (src/auth + src/common) reused by the lock -----------
# Constants.h is generated by the top-level CMakeLists.txt.
set(LOCK_CONSTANTS_INCLUDE_DIR "${CMAKE_BINARY_DIR}/src/common")

set(LOCK_AUTH_FILES
    "${LOCK_REPO_ROOT}/src/auth/Auth.cpp"
    "${LOCK_REPO_ROOT}/src/auth/Auth.h"
    "${LOCK_REPO_ROOT}/src/auth/AuthMessages.h"
    "${LOCK_REPO_ROOT}/src/auth/AuthPrompt.cpp"
    "${LOCK_REPO_ROOT}/src/auth/AuthPrompt.h"
    "${LOCK_REPO_ROOT}/src/auth/AuthRequest.cpp"
    "${LOCK_REPO_ROOT}/src/auth/AuthRequest.h"
    "${LOCK_REPO_ROOT}/src/common/MessageHandler.h"
    "${LOCK_REPO_ROOT}/src/common/SafeDataStream.cpp"
    "${LOCK_REPO_ROOT}/src/common/SafeDataStream.h"
)

# --- Lock screen (reuses SDDM::Auth in authentication-only mode) -------------
set(EXT_SESSION_LOCK_PROTOCOL
    "${WAYLAND_PROTOCOLS_DATADIR}/staging/ext-session-lock/ext-session-lock-v1.xml")
set(EXT_SESSION_LOCK_CLIENT_HEADER
    "${CMAKE_CURRENT_BINARY_DIR}/wayland-ext-session-lock-v1-client-protocol.h")
set(EXT_SESSION_LOCK_PROTOCOL_CODE
    "${CMAKE_CURRENT_BINARY_DIR}/wayland-ext-session-lock-v1-protocol.c")

add_custom_command(
    OUTPUT "${EXT_SESSION_LOCK_CLIENT_HEADER}"
    COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" --include-core-only client-header
            "${EXT_SESSION_LOCK_PROTOCOL}" "${EXT_SESSION_LOCK_CLIENT_HEADER}"
    DEPENDS "${EXT_SESSION_LOCK_PROTOCOL}"
    VERBATIM
)

add_custom_command(
    OUTPUT "${EXT_SESSION_LOCK_PROTOCOL_CODE}"
    COMMAND "${WAYLAND_SCANNER_EXECUTABLE}" --include-core-only public-code
            "${EXT_SESSION_LOCK_PROTOCOL}" "${EXT_SESSION_LOCK_PROTOCOL_CODE}"
    DEPENDS "${EXT_SESSION_LOCK_PROTOCOL}" "${EXT_SESSION_LOCK_CLIENT_HEADER}"
    VERBATIM
)

# The generated files belong to a target in this directory: CMake only runs a
# custom command for targets defined next to it, and with policy CMP0118 OLD
# (cmake_minimum_required 3.16) other directories would not even know the
# files are generated. gxde-lock-backend (src/lock-common) links this instead
# of listing them; a clean configure failed with "Cannot find source file".
add_library(gxde-lock-ext-session-lock-protocol STATIC
    "${EXT_SESSION_LOCK_CLIENT_HEADER}"
    "${EXT_SESSION_LOCK_PROTOCOL_CODE}"
)
target_include_directories(gxde-lock-ext-session-lock-protocol PUBLIC
    "${CMAKE_CURRENT_BINARY_DIR}"
)

# Shared authentication, user and D-Bus backend plus the QML-facing facade.
# The classic locker keeps its own copies for now; gxde-lock-neo links these
# libraries so both frontends stay in sync.
set(LOCK_CLASSIC_BINARY_DIR "${CMAKE_CURRENT_BINARY_DIR}")
add_subdirectory(
    "${LOCK_REPO_ROOT}/src/lock-common"
    "${CMAKE_CURRENT_BINARY_DIR}/lock-common"
)

file(GLOB NEO_LOCK_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/lock/*.cpp"
    "${CMAKE_CURRENT_SOURCE_DIR}/lock/dbus/*.cpp"
)
file(GLOB NEO_LOCK_HEADERS CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/lock/*.h"
    "${CMAKE_CURRENT_SOURCE_DIR}/lock/dbus/*.h"
)
add_executable(gxde-lock-classic
    ${NEO_LOCK_SOURCES}
    ${NEO_LOCK_HEADERS}
    ${NEO_COMMON_SOURCES}
    ${NEO_COMMON_HEADERS}
    ${NEO_VENDOR_SOURCES}
    ${NEO_VENDOR_HEADERS}
    ${LOCK_AUTH_FILES}
    "${LOCK_REPO_ROOT}/src/lock-common/lockstyle.cpp"
    "${LOCK_REPO_ROOT}/src/lock-common/lockstyle.h"
    "${LOCK_REPO_ROOT}/src/lock-common/lockstylemanager.cpp"
    "${LOCK_REPO_ROOT}/src/lock-common/lockstylemanager.h"
    "${LOCK_REPO_ROOT}/src/common/FingerprintProbe.cpp"
    "${LOCK_REPO_ROOT}/src/common/FingerprintProbe.h"
    "${LOCK_REPO_ROOT}/src/common/FingerprintSession.cpp"
    "${LOCK_REPO_ROOT}/src/common/FingerprintSession.h"
    "${LOCK_REPO_ROOT}/src/common/LockState.cpp"
    "${LOCK_REPO_ROOT}/src/common/LockState.h"
    "${LOCK_REPO_ROOT}/src/common/LogindSession.cpp"
    "${LOCK_REPO_ROOT}/src/common/LogindSession.h"
    lock/images.qrc
    lock/logintheme.qrc
    ${NEO_WIDGET_RESOURCES}
)
target_include_directories(gxde-lock-classic PRIVATE
    "${CMAKE_CURRENT_BINARY_DIR}"
    ${NEO_INCLUDE_DIRS}
    "${CMAKE_CURRENT_SOURCE_DIR}/lock"
    "${LOCK_REPO_ROOT}/src/auth"
    "${LOCK_REPO_ROOT}/src/common"
    "${LOCK_REPO_ROOT}/src/lock-common"
    "${LOCK_CONSTANTS_INCLUDE_DIR}"
)
target_link_libraries(gxde-lock-classic PRIVATE
    ${NEO_COMMON_LIBS}
    gxde-lock-ext-session-lock-protocol
    LayerShellQt::Interface
    Qt6::Qml
    Qt6::WaylandClientPrivate
)
install(TARGETS gxde-lock-classic DESTINATION "${CMAKE_INSTALL_BINDIR}")

# --- QML lock screen (gxde-lock-neo) -----------------------------------------
if(BUILD_NEO_LOCK)
    add_subdirectory(
        "${LOCK_REPO_ROOT}/src/lock-neo"
        "${CMAKE_CURRENT_BINARY_DIR}/lock-neo"
    )
endif()
